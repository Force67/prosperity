// The render thread records a draw after the walk resolved it, and by then the
// title has already refilled the descriptor tables and instance rings that draw
// points at: instanced props and skinned meshes read other objects' transforms
// and smear across the screen. Recording on the walk thread reads them in time
// (2026-10-23).
+DELTA_GPU_RENDER_THREAD=0

// The cockpit glass is one sided: drawn from inside without culling, its outer
// faces cover the windshield in opaque grey (fresnel of a back face).
+DELTA_GPU_CULL=1

// The v1.28 (60 fps) cockpit camera flips between two poses every frame once
// a frame takes more than ~0.06 s of game time, and motion blur smears the
// interior across them. Below 20 fps the race runs in slow motion instead.
+DELTA_PROCTIME_FRAME_CAP_MS=50
